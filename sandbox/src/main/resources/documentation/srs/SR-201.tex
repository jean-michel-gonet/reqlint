\begin{softwarerequirement}{SR-201}{Lazy Replenishment Calculation}
\paragraph{Links} \childof{ER-200}

Upon receiving a client request,
the server shall calculate the newly added tokens $\Delta T$ since the last request
using the formula~\ref{eq:lazy-replenishment-calculation},
where $R$ is the sustained refill rate in tokens per second.

\begin{equation}
    \Delta T = (t_{current} - t_{last}) \times R
    \label{eq:lazy-replenishment-calculation}
\end{equation}
\end{softwarerequirement}
